    function playBeep(type = 'success') {
      try {
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        osc.connect(gainNode);
        gainNode.connect(audioCtx.destination);

        if (type === 'success') {
          osc.frequency.setValueAtTime(587.33, audioCtx.currentTime); // D5
          osc.frequency.setValueAtTime(880, audioCtx.currentTime + 0.1); // A5
          gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime);
          gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.3);
        } else {
          // Error tone
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(220, audioCtx.currentTime);
          osc.frequency.setValueAtTime(160, audioCtx.currentTime + 0.15);
          gainNode.gain.setValueAtTime(0.2, audioCtx.currentTime);
          gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.35);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.35);
        }
      } catch (e) {
        // AudioContext restricted before gesture
      }
    }

    /* Distance Calculation (Haversine Formula) */
    function calculateDistanceInMeters(lat1, lon1, lat2, lon2) {
      const R = 6371e3; // metres
      const φ1 = lat1 * Math.PI / 180;
      const φ2 = lat2 * Math.PI / 180;
      const Δφ = (lat2 - lat1) * Math.PI / 180;
      const Δλ = (lon2 - lon1) * Math.PI / 180;

      const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                Math.cos(φ1) * Math.cos(φ2) *
                Math.sin(Δλ/2) * Math.sin(Δλ/2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
      return R * c;
    }

    function getCombinedDailyRecords() {
      const currentGross = (state.transactions || []).reduce((sum, tx) => sum + tx.total, 0);
      const currentCOGS = (state.transactions || []).reduce((sum, tx) => sum + (tx.cogs || 0), 0);

        date: state.currentDayDate,
        orderCount: (state.transactions || []).length,
        grossSales: currentGross,
        totalExpenses: currentExpenses,
        netProfit: currentGross - currentCOGS - currentExpenses

      const map = new Map();
      [liveDay, ...archives].forEach(r => { if (!map.has(r.date)) map.set(r.date, r); });
      return Array.from(map.values()).sort((a, b) => new Date(a.date) - new Date(b.date));
    }

    function renderDashboardPeriod() {
      const records = getCombinedDailyRecords();
      const todayDate = new Date(state.currentDayDate);
      let filteredRecords = [];

      if (activeDashboardPeriod === 'day') {
        filteredRecords = records.filter(r => r.date === state.currentDayDate);
      } else if (activeDashboardPeriod === 'week') {
        const d = new Date(todayDate); d.setDate(d.getDate() - 7);
        filteredRecords = records.filter(r => new Date(r.date) >= d);
      } else if (activeDashboardPeriod === 'month') {
        filteredRecords = records.filter(r => new Date(r.date).getFullYear() === todayDate.getFullYear() && new Date(r.date).getMonth() === todayDate.getMonth());
      } else if (activeDashboardPeriod === 'year') {
        filteredRecords = records.filter(r => new Date(r.date).getFullYear() === todayDate.getFullYear());
      }

      const totalGross = filteredRecords.reduce((s, r) => s + (r.grossSales||0), 0);
      const totalOrders = filteredRecords.reduce((s, r) => s + (r.orderCount||0), 0);
      const totalProfit = filteredRecords.reduce((s, r) => s + (r.netProfit||0), 0);
      const totalExp = filteredRecords.reduce((s, r) => s + (r.totalExpenses||0), 0);

      document.getElementById('kpi-gross-revenue').innerText = `RM ${totalGross.toFixed(2)}`;
      document.getElementById('kpi-order-count').innerText = `${totalOrders} completed orders`;
      document.getElementById('kpi-net-profit').innerText = `RM ${totalProfit.toFixed(2)}`;
      document.getElementById('kpi-total-expenses').innerText = `RM ${totalExp.toFixed(2)}`;
      document.getElementById('kpi-integrity-score').innerText = `${state.integrityScore.toFixed(1)}%`;

      renderDashboardChart(filteredRecords);
      renderPaymentDonutChart();
    }

    function renderPaymentDonutChart() {
      const canvas = document.getElementById('paymentDonutChart');
      if (!canvas || typeof Chart === 'undefined') return;
      if (paymentChartInstance) paymentChartInstance.destroy();

      let digitalTotal = 0;
      let cashTotal = 0;
      (state.transactions || []).forEach(t => {
        if (t.paymentMethod === 'Cash') cashTotal += t.total;
        else digitalTotal += t.total;
      });

      const total = digitalTotal + cashTotal;
      const digPct = total > 0 ? Math.round((digitalTotal / total) * 100) : 0;
      const cashPct = total > 0 ? 100 - digPct : 0;

      document.getElementById('tender-digital-pct').innerText = `${digPct}%`;
      document.getElementById('tender-cash-pct').innerText = `${cashPct}%`;

      const chartData = total > 0 ? [digitalTotal, cashTotal] : [1, 0];
      const chartColors = total > 0 ? ['#6366F1', '#10B981'] : ['#1E293B', '#1E293B'];

      paymentChartInstance = new Chart(canvas.getContext('2d'), {
        type: 'doughnut',
        data: {
          labels: ['Digital (QR/Card)', 'Cash'],
          datasets: [{
            data: chartData,
            backgroundColor: chartColors,
            borderColor: '#0B0F19',
            borderWidth: 3
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          cutout: '72%',
          plugins: { legend: { display: false } }
        }
      });
    }

    function openStoreQrModal() {

      document.getElementById('qr-modal-radius').innerText = `Within ${state.geofence.radiusMeters} meters`;

      // Compact token format to prevent qrcode.min.js buffer overflow (fits well within QR bit limits)
      const compactToken = btoa(`${state.currentDayDate}:${state.masterPin}`).slice(0, 10);
      const qrPayload = `APEX-ATT|${state.currentDayDate}|${state.geofence.latitude}|${state.geofence.longitude}|${state.geofence.radiusMeters}|${compactToken}`;

      const container = document.getElementById('qrcode-container');
      container.innerHTML = '';

      try {
        if (typeof QRCode !== 'undefined') {
          qrcodeInstance = new QRCode(container, {
            text: qrPayload,
            width: 190,
            height: 190,
            colorDark: "#0B0F19",
            colorLight: "#ffffff",
            correctLevel: QRCode.CorrectLevel.L
          });
        } else {
          throw new Error("QRCode library unavailable");
        }
      } catch (err) {
        console.warn("QRCodeJS render fallback:", err);
        // Fallback generator using standard Google Chart / QR image API
        const safeUrl = encodeURIComponent(qrPayload);
        container.innerHTML = `<img src="https://api.qrserver.com/v1/create-qr-code/?size=190x190&data=${safeUrl}" alt="Store QR Code" class="w-full h-full object-contain rounded-lg shadow-sm" onerror="this.onerror=null; this.parentElement.innerHTML='<div class=\'text-center p-4 text-xs font-mono text-slate-800 font-bold\'>APEX SECURE ATTENDANCE TOKEN<br><span class=\'text-[10px] text-slate-500\'>${state.currentDayDate}</span></div>';">`;
      }

      if (typeof lucide !== 'undefined') lucide.createIcons();
    }

    function toggleStaffShiftManual(staffId) {
      const staff = state.staff.find(s => s.id === staffId);
      if (!staff) return;

      const nowTimeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      if (!staff.clockedIn) {
        staff.clockedIn = true;
        staff.clockInTime = nowTimeStr;
        staff.clockInTimestamp = Date.now();
        staff.lastCoords = `${state.geofence.latitude.toFixed(4)}° N, ${state.geofence.longitude.toFixed(4)}° E (Owner Override)`;
        showToast(`${staff.name} clocked in manually by Owner.`);
      } else {
        staff.clockedIn = false;
        staff.hoursClocked += 4.0; // default standard session
        staff.clockInTime = null;
        staff.clockInTimestamp = null;
        staff.lastCoords = `Owner Clocked Out at ${nowTimeStr}`;
        showToast(`${staff.name} clocked out manually by Owner (+4.0 hrs).`);
      }
      saveState();
      renderStaffAndPayroll();
    }

    function submitSale() {
      const skuId = document.getElementById('pos-product-select').value;
      if (!skuId) {
        showToast('Please add products in the Inventory tab first.', true);
        return;
      }
      const qty = parseInt(document.getElementById('pos-qty-input').value) || 1;
      const tender = document.getElementById('pos-tender-select').value;
      const prod = state.products.find(p => p.id === skuId);

      if (!prod) {
        showToast('Product not found in catalog.', true);
        return;
      }
      if (prod.stock < qty) {
        showToast(`Insufficient stock! Only ${prod.stock} units remaining.`, true);
        return;
      }

      // Decrement stock
      prod.stock -= qty;

      const subtotal = prod.price * qty;
      const cogs = prod.cost * qty;
      const nowStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      state.transactions.unshift({
        invoiceNumber: `INV-${Date.now().toString().slice(-4)}`,
        time: nowStr,
        itemsSummary: `${prod.name} x${qty}`,
        paymentMethod: tender,
        total: subtotal,
        cogs: cogs,
        sst: 0
      });

      if (tender === 'Cash') {
        state.tillFloat += subtotal;
      }

      saveState();
      closePosModal();
      renderAllViews();
      showToast(`Sale completed! RM ${subtotal.toFixed(2)} received via ${tender}.`);
    }

    function runCashReconciliation() {
      const counted = parseFloat(document.getElementById('audit-cash-input').value);
      if (isNaN(counted)) {
        showToast('Please input the actual physical counted cash.', true);
        return;
      }

      const expected = state.tillFloat;
      const variance = counted - expected;
      const resBox = document.getElementById('audit-result-box');
      resBox.classList.remove('hidden');

      if (Math.abs(variance) < 0.05) {
        state.integrityScore = 100.0;
        resBox.innerHTML = `
          <div class="text-emerald-400 font-bold flex items-center gap-2">
            <i data-lucide="check-circle-2" class="w-4 h-4"></i> Perfect Till Match (0.00 Variance)
          </div>
          <p class="text-slate-300">Expected: RM ${expected.toFixed(2)} | Counted: RM ${counted.toFixed(2)}</p>
          <p class="text-[11px] text-slate-400 mt-1">Cash drawer integrity verified with zero physical leakage.</p>
        `;
        showToast('Reconciliation Complete: Zero Variance!');
      } else if (variance < 0) {
        state.integrityScore = Math.max(70.0, 100 - (Math.abs(variance) / (expected || 1) * 20));
        resBox.innerHTML = `
          <div class="text-rose-400 font-bold flex items-center gap-2">
            <i data-lucide="alert-octagon" class="w-4 h-4"></i> Cash Shortage Detected: -RM ${Math.abs(variance).toFixed(2)}
          </div>
          <p class="text-slate-300">Expected: RM ${expected.toFixed(2)} | Counted: RM ${counted.toFixed(2)}</p>
          <p class="text-[11px] text-rose-300/80 mt-1">Notice: Immediate audit log flagged for shift inspection.</p>
        `;
        showToast(`Shortage of RM ${Math.abs(variance).toFixed(2)} flagged!`, true);
      } else {
        resBox.innerHTML = `
          <div class="text-amber-400 font-bold flex items-center gap-2">
            <i data-lucide="alert-triangle" class="w-4 h-4"></i> Cash Surplus Found: +RM ${variance.toFixed(2)}
          </div>
          <p class="text-slate-300">Expected: RM ${expected.toFixed(2)} | Counted: RM ${counted.toFixed(2)}</p>
          <p class="text-[11px] text-slate-400 mt-1">Unlogged cash sale or change discrepancy detected.</p>
        `;
        showToast(`Surplus of RM ${variance.toFixed(2)} logged.`);
      }

      saveState();
      renderAllViews();
      if (typeof lucide !== 'undefined') lucide.createIcons();
    }

    function promptDeleteStaff(staffId) {
      const staff = state.staff.find(s => s.id === staffId);
      if (!staff) return;

      pendingDeleteStaffId = staffId;
      document.getElementById('delete-staff-target-name').innerText = staff.name;
      document.getElementById('modal-delete-staff').classList.remove('hidden');
      if (typeof lucide !== 'undefined') lucide.createIcons();
    }

    function closeDeleteStaffModal() {
      pendingDeleteStaffId = null;
      document.getElementById('modal-delete-staff').classList.add('hidden');
    }

    function executeDeleteStaff() {
      if (!pendingDeleteStaffId) return;

      const idx = state.staff.findIndex(s => s.id === pendingDeleteStaffId);
      if (idx !== -1) {
        const removed = state.staff.splice(idx, 1)[0];
        saveState();
        renderStaffAndPayroll();
        showToast(`Staff member "${removed.name}" removed from store.`);
      }

      closeDeleteStaffModal();
    }

    function useCurrentLocationForGeofence() {
      if (!navigator.geolocation) {
        showToast('Geolocation is not supported by your browser.', true);
        return;
      }
      navigator.geolocation.getCurrentPosition(
        (pos) => {
          state.geofence.latitude = pos.coords.latitude;
          state.geofence.longitude = pos.coords.longitude;
          document.getElementById('setting-store-lat').value = pos.coords.latitude.toFixed(6);
          document.getElementById('setting-store-lng').value = pos.coords.longitude.toFixed(6);
          showToast(`Captured Current GPS: ${pos.coords.latitude.toFixed(4)}°, ${pos.coords.longitude.toFixed(4)}°`);
        },
        (err) => {
          showToast(`GPS Error: ${err.message}`, true);
        }
      );
    }

    function exportDashboardCsv() {
      const records = getCombinedDailyRecords();
      let csv = "Date,Completed_Orders,Gross_Sales_MYR,Expenses_MYR,Net_Profit_MYR\n";
      records.forEach(r => {
        csv += `${r.date},${r.orderCount},${r.grossSales.toFixed(2)},${r.totalExpenses.toFixed(2)},${r.netProfit.toFixed(2)}\n`;
      });
      const blob = new Blob([csv], { type: 'text/csv' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `Apex_Sales_Report_${state.currentDayDate}.csv`;
      a.click();
      showToast('Sales CSV Exported.');
    }

    function exportPayrollCsv() {
      let csv = "Staff_ID,Staff_Name,Role,Hourly_Wage,Hours_Worked,Gross_Pay,EPF_11_Pct,SOCSO_EIS,Net_Payable\n";
      state.staff.forEach(s => {
        const gross = s.wage * s.hoursClocked;
        const epf = gross * 0.11;
        const socso = gross * 0.005;
        const net = gross - epf - socso;
        csv += `${s.id},"${s.name}","${s.role}",${s.wage.toFixed(2)},${s.hoursClocked.toFixed(2)},${gross.toFixed(2)},${epf.toFixed(2)},${socso.toFixed(2)},${net.toFixed(2)}\n`;
      });
      const blob = new Blob([csv], { type: 'text/csv' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `Apex_Autopay_Payroll_${state.currentDayDate}.csv`;
      a.click();
      showToast('Autopay Payroll CSV Exported.');
    }
